Freeze both listed enemies while you deal with King.

apocalypse.onInteract(apocTypes.Room.R27, apocTypes.Station.Primary, function () {
    let enemies: Sprite[] = apocalypse.roster()
    for (let enemy of enemies) {
        apocalypse.hold(enemy)
    }
})

## Right altar: clear the Shields

Clear all of King's shield casts.

![Blocks and variable setup for Right altar in room 27](https://raw.githubusercontent.com/mrbrackebusch-code/apocalyptic-lists/main/assets/instructions/v0.6.0/R27-secondary-menu.svg)

![Complete Right altar code for room 27, with explanations beside the relevant blocks](https://raw.githubusercontent.com/mrbrackebusch-code/apocalyptic-lists/main/assets/instructions/v0.6.0/R27-secondary-assembled.svg)

#### ~ tutorialhint

```blocks
apocalypse.onInteract(apocTypes.Room.R27, apocTypes.Station.Secondary, function () {
    let enemyMoves: string[] = apocalypse.moves(apocTypes.Side.Enemy)
    for (let index = 0; index <= enemyMoves.length - 1; index++) {
        if (enemyMoves[index] == "Shield") {
            enemyMoves[index] = "Wait"
        }
    }
    apocalypse.setMoves(apocTypes.Side.Enemy, enemyMoves)
})
```
